容易考虑到二分答案。

假设现在要验证答案为 $X$ 的正确性，那么一个学生与他印象不前 $X$ 好的学生不能在一组。这个东西的形式很像 `2-sat`，于是考虑用 `2-sat` 以验证正确性。

具体的，假设学生 $x$ 对学生 $y$ 的印象不是他眼中前 $X$ 好的，不妨设 $x$ 和 $y$ 都不能选择第一个老师，那么当 $x$ 选择第一个老师时，$y$ 不能选第一个老师，当  $y$ 选择第一个老师时，$x$ 不能选第一个老师。

#include <bits/stdc++.h>
using namespace std;

#define ll long long 
#define the_end return 0;

const int M=4e5;
const int N=1000;

int n;
int a[N+10][N+10],w[N+10][2];
vector<int> G[M+10];

int dfn[M+10],low[M+10];
int Scc[M+10],sccnt,tot;
int stk[M+10],tal;

inline void tarjan(int u){
	dfn[u]=low[u]=++tot;
	stk[++tal]=u;

	for(auto v:G[u]){
		if(!dfn[v]){
			tarjan(v);
			low[u]=min(low[u],low[v]);
		}
		else if(!Scc[v]){
			low[u]=min(low[u],dfn[v]);
		}
	}

	if(low[u]==dfn[u]){
		sccnt++;
		while(true){
			int x=stk[tal];
			tal--;
			Scc[x]=sccnt;
			if(u==x) break;
		}
	}
}

bool check(int mid){

    for(int i=1;i<=n*2;i++){
        G[i].clear();
        dfn[i]=Scc[i]=low[i]=0;
    }
    sccnt=tot=tal=0;

    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            if(i==j) continue;
            if(a[i][j]<=mid) continue;

            if(w[i][0]==w[j][0]){
                G[i].push_back(j+n);
                G[j].push_back(i+n);
            }
            else if(w[i][0]==w[j][1]){
                G[i].push_back(j);
                G[j+n].push_back(i+n);
            }

            if(w[i][1]==w[j][0]){
                G[i+n].push_back(j+n);
                G[j].push_back(i);
            }
            else if(w[i][1]==w[j][1]){
                G[i+n].push_back(j);
                G[j+n].push_back(i);
            }
        }
    }

    for(int i=1;i<=2*n;i++){
        if(!dfn[i]){
            tarjan(i);
        }
        if(mid!=3) continue;
        // cout<<i<<":";
        // for(auto u:G[i]){
        //     cout<<u<<" ";
        // }
        // cout<<endl;
    }

    for(int i=1;i<=n;i++){
        //if(mid==3)cout<<Scc[i]<<" "<<Scc[i+n]<<endl;
        if(Scc[i]==Scc[i+n]){
            return false;
        }
    }

    return true;
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(0),cout.tie(0);

    cin>>n;
    for(int i=1;i<=n;i++){
        int x,y=0;
        cin>>x;
        for(auto k:{0,1,2}){
            if(k==x) continue;
            w[i][y++]=k;
        }
        for(int j=1;j<=n-1;j++){
            int z;
            cin>>z;
            a[i][z]=j;
        }
        //cout<<w[i][0]<<" "<<w[i][1]<<endl;
    }

    int L=0,R=n,ans=0;
    while(L<=R){
        int mid=L+R>>1;
        if(check(mid)){
            ans=mid;
            R=mid-1;
        }
        else {
            L=mid+1;
        }
    }

    cout<<ans<<endl;
	
    the_end
}
