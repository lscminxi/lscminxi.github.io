考虑到在一个 DAG 中，只要有入度的点都可以选。具体的，将要选的点按照拓扑序从后往前选，就可以保证所有有入度的点都可以选到。

于是考虑将图构造成 DAG，考虑将图中的强连通分量缩点。对于一个缩完的点，如果这个点有入度或有自环，那么里面的点就全都可以选，否则就必须留下一个点。

证明很容易，考虑缩点后图中的一个强连通分量，如果这个强连通分量有入度，就将在这个强连通分量中留下一个有从其他强连通分量连进来的点，最后将这个点删掉。自环同理，可以考虑将有自环的那个点最后删除。而如果不是上述两种情况，那么就一定会留下一个点。

于是贪心的考虑将不能全删去的强连通分量留下点权最少的点，然后选择可以删的点中最大的 $k$ 个点即可。

#include <bits/stdc++.h>
using namespace std;

#define ll long long 
#define the_end return 0;

const int N=5e6;

ll n,m,K,vis[N+10];
ll W[N+10];
vector<int> G[N+10];
int low[N+10],dfn[N+10],scc[N+10],Mi[N+10],tim;
int stk[N+10],tt,scc_cnt,du[N+10];
ll b[N+10],bn,ans;

inline void dfs(ll u,ll fa){

	dfn[u]=low[u]=++tim;
	stk[++tt]=u;
	for(auto v:G[u]){
		if(!dfn[v]){
			dfs(v,u);
			low[u]=min(low[u],low[v]);
		}
		else if(!scc[v]){
			low[u]=min(low[u],dfn[v]);
		}
	}

	if(dfn[u]==low[u]){
		scc_cnt++;
		int X=0;
		Mi[scc_cnt]=0;
		do{
			X=stk[tt--];
			if(W[Mi[scc_cnt]]>=W[X]){
				Mi[scc_cnt]=X;
			}
			scc[X]=scc_cnt;	
		}while(X!=u);
	}
}

bool cmp(ll x,ll y){
	return x>y;
}

int main(){

    ios::sync_with_stdio(false);
    cin.tie(0),cout.tie(0);

	cin>>n>>m>>K;
	W[0]=LLONG_MAX;
	for(int i=1;i<=n;i++){
		cin>>W[i];
	}

	for(int i=1;i<=m;i++){
		int u,v;
		cin>>u>>v;
		G[u].push_back(v);
	}

	for(int i=1;i<=n;i++){
		if(!dfn[i]){
			dfs(i,0);
		}
	}

	for(int u=1;u<=n;u++){
		for(auto v:G[u]){
			if(scc[u]!=scc[v]){
				du[scc[v]]=1;
			}
			if(u==v){
				du[scc[v]]=1;
			}
		}
	}

	for(int i=1;i<=n;i++){
		if(du[scc[i]] || Mi[scc[i]]!=i){
			b[++bn]=W[i];
		}
	}

	sort(b+1,b+1+bn,cmp);

	for(int i=1;i<=min(bn,K);i++){
		ans+=b[i];
	}
	cout<<ans;
	
	the_end
}
